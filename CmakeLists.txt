# --------------------------------------------------
# Minimum CMake Version
# --------------------------------------------------

cmake_minimum_required(VERSION 3.20)

# --------------------------------------------------
# Project Name
# --------------------------------------------------

project(CustomGameServer)

# --------------------------------------------------
# C++ Standard
# --------------------------------------------------

# Use C++17
set(CMAKE_CXX_STANDARD 17)

# Force compiler to use required standard
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# --------------------------------------------------
# GLOBAL DEFINITIONS
# --------------------------------------------------

# Disable Boost automatic library linking
# (important for MSVC)
add_definitions(
    -DBOOST_ALL_NO_LIB
)

# Windows-specific definitions
if(WIN32)

    # Target Windows 10 APIs
    add_definitions(
        -D_WIN32_WINNT=0x0A00
    )

endif()

# --------------------------------------------------
# BOOST INCLUDE PATH
# --------------------------------------------------

# Root Boost directory
set(BOOST_ROOT "C:/Boost")

# Include directories
include_directories(

    include

    C:/Boost/include/boost-1_90
)

# --------------------------------------------------
# SOURCE FILES
# --------------------------------------------------

set(SOURCES

    # Entry Point
    src/main.cpp

    # Boost JSON Implementation Unit
    src/boost_json.cpp

    # Network
    src/network/server.cpp
    src/network/realtimeServer.cpp
    src/network/clientConnection.cpp
    src/network/connectionManager.cpp
    src/network/message.cpp

    # Lobby System
    src/lobby/lobby.cpp
    src/lobby/lobbyManager.cpp

    # Routes
    src/routes/lobbyRoutes.cpp
    src/routes/sessionRoutes.cpp

    # Router
    src/router/router.cpp

    # Session System
    src/session/session.cpp
    src/session/sessionManager.cpp

    # Middleware
    src/middleware/authMiddleware.cpp
)

# --------------------------------------------------
# CREATE EXECUTABLE
# --------------------------------------------------

add_executable(
    app
    ${SOURCES}
)

# --------------------------------------------------
# PLATFORM-SPECIFIC LIBRARIES
# --------------------------------------------------

# Windows socket library
if(WIN32)

    target_link_libraries(

        app

        ws2_32
    )

endif()
